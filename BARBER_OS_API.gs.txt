/************ INSTALADOR ************/

function instalarBarberOSLite() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const ui = SpreadsheetApp.getUi();

  crearHojaConfig_(ss, false);
  crearHojaClientes_(ss, false);
  crearHojaCitas_(ss, false);
  crearHojaServicios_(ss, false);
  crearHojaBarberos_(ss, false);
  crearHojaMensajes_(ss, false);
  crearHojaLog_(ss, false);

  migrarEsquema_();   // agrega columnas nuevas a CITAS si vienes de una versión anterior

  const cfgActual = cargarConfig_();
  if (!String(cfgActual["PIN_ACCESO"] || "").trim()) {
    guardarEnConfig_(ss, "PIN_ACCESO", generarPin_());
  }

  crearTriggersBarberOS_();

  const tz = Session.getScriptTimeZone();
  const bloqueTz = (tz === "America/Mexico_City") ? "" :
    `⚠️ Zona horaria del proyecto: ${tz}\nCámbiala a America/Mexico_City en Configuración del proyecto.\n\n`;

  const url = urlWebApp_();
  const bloqueUrl = url
    ? "Tus páginas ya están publicadas. Menú 📋 'Ver link — Panel' te da el enlace (incluye el PIN).\n\n"
    : "IMPORTANTE — Publica la Web App:\nImplementar → Nueva implementación → Aplicación web\n(Ejecutar como: Yo · Acceso: Cualquier persona)\n\n";

  ui.alert("✅ BARBER OS Lite v4.0 instalado",
    "Sistema listo.\n\n" + bloqueTz + bloqueUrl +
    "Llena CONFIG (nombre, teléfono, logo), agrega tus barberos en la hoja BARBEROS " +
    "y tus servicios/precios en SERVICIOS.",
    ui.ButtonSet.OK);
}

function reinstalarBarberOSLite() {
  const ui = SpreadsheetApp.getUi();
  const r  = ui.alert("⚠️ ADVERTENCIA",
    "Reinstalará CONFIG, SERVICIOS, BARBEROS y MENSAJES.\nCLIENTES, CITAS y LOG no se tocan.\n\n¿Continuar?",
    ui.ButtonSet.YES_NO);
  if (r !== ui.Button.YES) return;
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  crearHojaConfig_(ss, true);
  crearHojaServicios_(ss, true);
  crearHojaBarberos_(ss, true);
  crearHojaMensajes_(ss, true);
  migrarEsquema_();
  crearTriggersBarberOS_();
  ui.alert("✅ Configuración reinstalada.");
}

/************ LINK DEL PANEL ************/

function urlWebApp_() {
  try { return ScriptApp.getService().getUrl() || ""; } catch (e) { return ""; }
}

function generarPin_() {
  return Utilities.getUuid().replace(/-/g, "").substring(0, 8);
}

function linkConParametros_(page) {
  const url = urlWebApp_();
  if (!url) return "";
  let link = url + "?page=" + page;
  const pin = String(cargarConfig_()["PIN_ACCESO"] || "").trim();
  if (pin) link += "&key=" + encodeURIComponent(pin);
  return link;
}

function mostrarLinkPanel() {
  const ui   = SpreadsheetApp.getUi();
  const link = linkConParametros_("panel");
  const pin  = String(cargarConfig_()["PIN_ACCESO"] || "").trim();
  if (!link) {
    ui.alert("⚠️ Falta publicar la Web App",
      "Apps Script → Implementar → Nueva implementación → Aplicación web\n(Ejecutar como: Yo · Acceso: Cualquier persona).",
      ui.ButtonSet.OK);
    return;

    "\n\nO comparte el link sin el código y da el PIN aparte:\nPIN de acceso: " + (pin || "(sin PIN)"),
    ui.ButtonSet.OK);
}

/************ WEB APP — RUTEO ************/

function doGet(e) {
  const params = (e && e.parameter) || {};

  // ===== API JSON para el front alojado en GitHub Pages =====
  if (params.action) return apiJson_(params);

  const config = cargarConfig_();
  const titulo = (config["NOMBRE_BARBERIA"] || "Barber Studio");

  const pin = String(config["PIN_ACCESO"] || "").trim();
  if (pin && String(params.key || "") !== pin) {
    const intentoFallido = String(params.key || "") !== "";
    return HtmlService.createHtmlOutput(htmlAccesoRestringido_(config, "panel", intentoFallido))
      .setTitle(titulo + " — Acceso")
      .addMetaTag("viewport", "width=device-width, initial-scale=1")
      .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
  }

  return HtmlService.createHtmlOutput(htmlApp_(config))
    .setTitle(titulo + " — Panel")
    .addMetaTag("viewport", "width=device-width, initial-scale=1")
    .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
}

/************ API JSON (para el front de GitHub Pages) ************/

function apiJson_(p) {
  const salida = function (obj) {
    return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
  };
  try {
    const config = cargarConfig_();
    const pin = String(config["PIN_ACCESO"] || "").trim();
    const accion = String(p.action || "");

    // 'ping' es público (sin llave) para traer la marca antes de entrar
    if (accion === "ping") {
      return salida({ ok: true, barberia: config["NOMBRE_BARBERIA"] || "Barber Studio",
        acento: config["COLOR_ACENTO"] || "#1A1A1A", logo: config["LOGO_URL"] || "" });
    }

    // El resto exige la llave
    if (pin && String(p.key || "") !== pin) return salida({ ok: false, error: "auth" });

    switch (accion) {
      case "listas":    return salida({ ok: true, data: appListas() });
      case "dia":       return salida({ ok: true, data: appCargarDia(String(p.fecha || "")) });
      case "confirmar": return salida(appConfirmar(String(p.id || ""), String(p.accion || "")));
      case "agendar":   return salida(appAgendar({
        modo: p.modo, nombre: p.nombre, telefono: p.telefono, email: p.email,
        barbero: p.barbero, servicio: p.servicio, fecha: p.fecha, horaIni: p.horaIni
      }));
      default: return salida({ ok: false, error: "accion_desconocida" });
    }
  } catch (err) {
    return salida({ ok: false, error: String(err && err.message || err) });
  }
}

function appCargarDia(fechaISO) {
  const hoja = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(HOJAS.CITAS);
  const out = { citas: [], stats: { total: 0, done: 0, cash: 0 } };
  if (!hoja) return out;
  const data = hoja.getDataRange().getValues();
  if (data.length <= 1) return out;
  const headers = data[0];
  const objetivo = limpiarFecha_(fechaISO);
  if (!objetivo) return out;

  for (let i = 1; i < data.length; i++) {
    const fila = data[i];
    const fecha = limpiarFecha_(getValor_(fila, headers, "FECHA_CITA"));
    if (!fecha || fecha.getTime() !== objetivo.getTime()) continue;

    const estadoRaw = String(getValor_(fila, headers, "ESTADO_CITA")).toUpperCase();
    let estado = "pend";
    if (estadoRaw === "COMPLETADA") estado = "done";
    else if (estadoRaw === "CANCELADA" || estadoRaw === "REAGENDADA") estado = "canc";

    const serv = getValor_(fila, headers, "SERVICIO");
    if (estado === "done") { out.stats.done++; out.stats.cash += obtenerPrecioServicio_(serv); }
    if (estado !== "canc") out.stats.total++;

    out.citas.push({
      id:     String(getValor_(fila, headers, "ID_CITA")),
      ini:    formatearHora_(getValor_(fila, headers, "HORA_CITA")),
      fin:    formatearHora_(getValor_(fila, headers, "HORA_FIN")),
      n:      getValor_(fila, headers, "NOMBRE_CLIENTE"),
      s:      serv,
      b:      getValor_(fila, headers, "BARBERO"),
      t:      getValor_(fila, headers, "TELEFONO"),
      estado: estado,
      _o:     (function(){ const h=String(getValor_(fila, headers, "HORA_CITA")).split(":"); return (Number(h[0])||0)*60+(Number(h[1])||0); })()
    });
  }
  out.citas.sort((a, b) => a._o - b._o);
  return out;
}

function appConfirmar(idCita, accion) {
  try {
    const hoja = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(HOJAS.CITAS);
    if (!hoja) return { ok: false };
    const data = hoja.getDataRange().getValues();
    const headers = data[0];
    const config = cargarConfig_();
    let fila = -1;
    for (let i = 1; i < data.length; i++) {
      if (String(getValor_(data[i], headers, "ID_CITA")) === String(idCita)) { fila = i + 1; break; }
    }
    if (fila === -1) return { ok: false };

    if (accion === "done") {
      setValor_(hoja, headers, fila, "ESTADO_CITA", "COMPLETADA");
      setValor_(hoja, headers, fila, "FECHA_COMPLETADA", new Date());
      registrarOActualizarCliente_(data[fila - 1], headers, config, true);
      registrarLog_("PANEL_COMPLETADA", getValor_(data[fila - 1], headers, "NOMBRE_CLIENTE"), idCita, "OK");
    } else if (accion === "canc") {
      setValor_(hoja, headers, fila, "ESTADO_CITA", "CANCELADA");
      registrarLog_("PANEL_CANCELADA", getValor_(data[fila - 1], headers, "NOMBRE_CLIENTE"), idCita, "OK");
    }
    return { ok: true };
  } catch (e) {
    registrarLog_("ERROR_PANEL", "", idCita, e.message);
    return { ok: false, mensaje: e.message };
  }
}

function appAgendar(d) {
  try {
    d = d || {};
    const nombre   = String(d.nombre   || "").trim();
    const telefono = String(d.telefono || "").trim();
    const email    = String(d.email    || "").trim();
    const barbero  = String(d.barbero  || "").trim();
    const servicio = String(d.servicio || "").trim();
    const fechaRaw = String(d.fecha    || "").trim();
    const horaIni  = String(d.horaIni  || "").trim();

    if (!nombre || !telefono || !servicio || !fechaRaw || !horaIni) {
      return { ok: false, mensaje: "Faltan datos: revisa cliente, servicio, fecha y hora." };
    }
    const fechaDate = limpiarFecha_(fechaRaw);
    if (!fechaDate) return { ok: false, mensaje: "Fecha inválida." };

    const config  = cargarConfig_();
    const ss      = SpreadsheetApp.getActiveSpreadsheet();
    const hoja    = ss.getSheetByName(HOJAS.CITAS);
    if (!hoja) return { ok: false, mensaje: "Hoja CITAS no encontrada." };

    const telNorm = normalizarTelefono_(telefono, config) || telefono;
    const horaFin = calcularHoraFin_(horaIni, servicio);

    // Cliente
    upsertClienteBasico_(nombre, telNorm, email);

    // Confirmación al cliente (marca enviado solo si de verdad salió)
    const plantillas = cargarPlantillas_();
    const vars = { NOMBRE: nombre, FECHA: formatearFechaAmable_(fechaDate), HORA: formatearHoraAmable_(horaIni), SERVICIO: servicio };
    const enviado = enviarMensajeCliente_(telNorm, email,
      construirMensajeDesdeCache_(plantillas, "CONFIRMACION_CITA", vars, config), "CONFIRMACION_CITA", config, vars);

    appendFila_(hoja, {
      ID_CITA: crearId_("CITA"), FECHA_REGISTRO: new Date(), NOMBRE_CLIENTE: nombre,
      TELEFONO: telNorm, EMAIL_CLIENTE: email, SERVICIO: servicio, BARBERO: barbero,
      FECHA_CITA: fechaDate, HORA_CITA: horaIni, HORA_FIN: horaFin, ESTADO_CITA: "AGENDADA",
      MENSAJE_CONFIRMACION_ENVIADO: enviado ? "SÍ" : ""
    });

    registrarLog_("PANEL_AGENDADA", nombre, `${servicio} | ${fechaRaw} ${horaIni}-${horaFin} | ${barbero}`, "OK");
    return { ok: true, mensaje: "Cita agendada." };
  } catch (e) {
    registrarLog_("ERROR_AGENDAMIENTO", "", "", e.message);
    return { ok: false, mensaje: "Error: " + e.message };
  }
}

function getClientesLista_() {
  const hoja = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(HOJAS.CLIENTES);
  if (!hoja) return [];
  const data = hoja.getDataRange().getValues();
  if (data.length <= 1) return [];
  const h = data[0]; const cfg = cargarConfig_();
  const map = {}; const orden = [];
  for (let i = 1; i < data.length; i++) {
    if (String(getValor_(data[i], h, "ESTADO_CLIENTE")).toUpperCase() === "INACTIVO") continue;
    const n = getValor_(data[i], h, "NOMBRE"); const t = getValor_(data[i], h, "TELEFONO");
    if (!n || !t) continue;
    const key = telClave_(t, cfg);                    // colapsa el mismo número en distintos formatos
    const m = String(getValor_(data[i], h, "EMAIL_CLIENTE") || "");
    if (!map[key]) { map[key] = { n: String(n), t: String(t), m: m }; orden.push(key); }
    else if (!map[key].m && m) { map[key].m = m; }     // completa el email si faltaba
  }
  const lista = orden.map(k => map[k]);
  lista.sort((a, b) => a.n.localeCompare(b.n));
  return lista;
}

function getBarberos_() {
  const hoja = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(HOJAS.BARBEROS);
  const lista = [];
  if (hoja) {
    const data = hoja.getDataRange().getValues();
    for (let i = 1; i < data.length; i++) {
      if (data[i][0] && String(data[i][1]).toUpperCase().indexOf("S") === 0) lista.push(String(data[i][0]).trim());
    }
  }
  if (lista.length === 0) {
    const nb = String(cargarConfig_()["NOMBRE_BARBERO"] || "").trim();
    lista.push(nb || "Barbero");
  }
  return lista;
}

function getServiciosDetalle_() {
  const hoja = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(HOJAS.SERVICIOS);
  const lista = [];
  if (hoja) {
    const data = hoja.getDataRange().getValues();
    for (let i = 1; i < data.length; i++) {
      if (!data[i][0]) continue;
      if (String(data[i][3]).toUpperCase().indexOf("S") !== 0) continue;
      lista.push({ nombre: String(data[i][0]), min: Number(data[i][1]) || 45, precio: Number(data[i][2]) || 0 });
    }
  }
  if (lista.length === 0) lista.push({ nombre: "Corte", min: 45, precio: 0 });
  return lista;
}

function obtenerDuracionServicio_(servicio) {
  const hoja = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(HOJAS.SERVICIOS);
  if (!hoja) return 45;
  const data = hoja.getDataRange().getValues();
  for (let i = 1; i < data.length; i++) {
    if (String(data[i][0]).toLowerCase() === String(servicio).toLowerCase()) return Number(data[i][1]) || 45;
  }
  return 45;
}

function obtenerPrecioServicio_(servicio) {
  const hoja = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(HOJAS.SERVICIOS);
  if (!hoja) return 0;
  const data = hoja.getDataRange().getValues();
  for (let i = 1; i < data.length; i++) {
    if (String(data[i][0]).toLowerCase() === String(servicio).toLowerCase()) return Number(data[i][2]) || 0;
  }
  return 0;
}

function calcularHoraFin_(horaIni, servicio) {
  const min = obtenerDuracionServicio_(servicio);
  const p = String(horaIni).split(":");
  const d = new Date(2000, 0, 1, Number(p[0]) || 0, Number(p[1]) || 0);
  const f = new Date(d.getTime() + min * 60000);
  return ("0" + f.getHours()).slice(-2) + ":" + ("0" + f.getMinutes()).slice(-2);
}

function upsertClienteBasico_(nombre, telefono, email) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const hoja = ss.getSheetByName(HOJAS.CLIENTES);
  if (!hoja) return;
  const data = hoja.getDataRange().getValues();
  const h = data[0]; const cfg = cargarConfig_(); const key = telClave_(telefono, cfg);
  const iTel = h.indexOf("TELEFONO");
  for (let i = 1; i < data.length; i++) {
    if (telClave_(data[i][iTel], cfg) === key) {
      setValor_(hoja, h, i + 1, "NOMBRE", nombre);
      setValor_(hoja, h, i + 1, "TELEFONO", telefono);   // normaliza el guardado
      if (email) setValor_(hoja, h, i + 1, "EMAIL_CLIENTE", email);
      setValor_(hoja, h, i + 1, "ESTADO_CLIENTE", "ACTIVO");
      return;
    }
  }
  appendFila_(hoja, {
    ID_CLIENTE: crearId_("CLI"), NOMBRE: nombre, TELEFONO: telefono,
    EMAIL_CLIENTE: email || "", TOTAL_VISITAS: 0, ESTADO_CLIENTE: "ACTIVO"
  });
}

function crearOReiniciarHoja_(ss, nombre, headers, datos, forzar) {
  let hoja = ss.getSheetByName(nombre);
  if (!hoja) { hoja = ss.insertSheet(nombre); }
  else { if (!forzar && hoja.getLastRow() > 1) { Logger.log(`[SKIP] "${nombre}" ya tiene datos.`); return; } hoja.clear(); }
  hoja.getRange(1, 1, 1, headers.length).setValues([headers]);
  if (datos.length > 0) hoja.getRange(2, 1, datos.length, headers.length).setValues(datos);
  hoja.setFrozenRows(1);
  hoja.autoResizeColumns(1, headers.length);
  hoja.getRange(1, 1, 1, headers.length).setFontWeight("bold").setBackground("#111111").setFontColor("#ffffff").setHorizontalAlignment("center");
  hoja.getDataRange().setVerticalAlignment("middle");
}

function migrarEsquema_() {
  const hoja = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(HOJAS.CITAS);
  if (!hoja) return;
  const headers = hoja.getRange(1, 1, 1, Math.max(1, hoja.getLastColumn())).getValues()[0].map(v => String(v).trim());
  ["BARBERO", "HORA_FIN", "FECHA_COMPLETADA"].forEach(function (col) {
    if (headers.indexOf(col) === -1) {
      const c = hoja.getLastColumn() + 1;
      hoja.getRange(1, c).setValue(col).setFontWeight("bold").setBackground("#111111").setFontColor("#ffffff");
      headers.push(col);
    }
  });
}

function guardarEnConfig_(ss, clave, valor) {
  const hoja = ss.getSheetByName(HOJAS.CONFIG);
  if (!hoja) return;
  const data = hoja.getDataRange().getValues();
  for (let i = 1; i < data.length; i++) {
    if (String(data[i][0]).trim() === clave) { hoja.getRange(i + 1, 2).setValue(valor); return; }
  }
  hoja.appendRow([clave, valor, "Generado automáticamente"]);
}

function sistemaActivo_(config) {
  const cfg = config || cargarConfig_();
  const v = String(cfg["SISTEMA_ACTIVO"] == null ? "SÍ" : cfg["SISTEMA_ACTIVO"]).trim().toUpperCase();
  return v !== "NO" && v !== "PAUSADO" && v !== "OFF";
}

/************ PROCESAR CITAS (respaldo horario, sin Calendar) ************/

function procesarCitasPendientes() {
  const hoja = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(HOJAS.CITAS);
  if (!hoja) return;
  const config = cargarConfig_();
  const plantillas = cargarPlantillas_();
  const data = hoja.getDataRange().getValues();
  if (data.length <= 1) return;
  const headers = data[0];

  for (let i = 1; i < data.length; i++) {
    const fila = data[i]; const row = i + 1;
    const nombre = getValor_(fila, headers, "NOMBRE_CLIENTE");
    const telefono = getValor_(fila, headers, "TELEFONO");
    const servicio = getValor_(fila, headers, "SERVICIO");
    const fechaCita = getValor_(fila, headers, "FECHA_CITA");
    const horaCita = getValor_(fila, headers, "HORA_CITA");
    if (!nombre || !telefono || !servicio || !fechaCita || !horaCita) continue;
    if (esCitaCerrada_(getValor_(fila, headers, "ESTADO_CITA"))) continue;

    if (getValor_(fila, headers, "MENSAJE_CONFIRMACION_ENVIADO") !== "SÍ") {
      const vars = { NOMBRE: nombre, FECHA: formatearFechaAmable_(fechaCita), HORA: formatearHoraAmable_(horaCita), SERVICIO: servicio };
      const msg = construirMensajeDesdeCache_(plantillas, "CONFIRMACION_CITA", vars, config);
      if (enviarMensajeCliente_(telefono, getValor_(fila, headers, "EMAIL_CLIENTE"), msg, "CONFIRMACION_CITA", config, vars)) {
        setValor_(hoja, headers, row, "MENSAJE_CONFIRMACION_ENVIADO", "SÍ");
      }
    }
  }
}

/************ CLIENTES (visitas) ************/

function registrarOActualizarCliente_(fila, headers, config, incrementarVisita) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const hoja = ss.getSheetByName(HOJAS.CLIENTES);
  if (!hoja) return;
  const data = hoja.getDataRange().getValues();
  const h = data[0];
  const nombre = getValor_(fila, headers, "NOMBRE_CLIENTE");
  const telefono = getValor_(fila, headers, "TELEFONO");
  const email = getValor_(fila, headers, "EMAIL_CLIENTE");
  const fechaCita = getValor_(fila, headers, "FECHA_CITA");
  const notas = getValor_(fila, headers, "NOTAS");
  if (!nombre || !telefono) return;

  let filaCliente = -1;
  const iTel = h.indexOf("TELEFONO"); const key = telClave_(telefono, config);
  for (let i = 1; i < data.length; i++) {
    if (telClave_(data[i][iTel], config) === key) { filaCliente = i + 1; break; }
  }
  const dias = Number(config["DIAS_RECORDATORIO_CORTE"]) || 21;
  let proximo = "";
  if (fechaCita) { const p = new Date(fechaCita); if (!isNaN(p.getTime())) { p.setDate(p.getDate() + dias); proximo = p; } }

  if (filaCliente === -1) {
    appendFila_(hoja, {
      ID_CLIENTE: crearId_("CLI"), NOMBRE: nombre, TELEFONO: telefono, EMAIL_CLIENTE: email || "",
      ULTIMA_VISITA: incrementarVisita ? fechaCita : "", PROXIMO_RECORDATORIO_CORTE: incrementarVisita ? proximo : "",
      TOTAL_VISITAS: incrementarVisita ? 1 : 0, ESTADO_CLIENTE: "ACTIVO", NOTAS: notas || ""
    });
    return;
  }
  setValor_(hoja, h, filaCliente, "NOMBRE", nombre);
  if (email) setValor_(hoja, h, filaCliente, "EMAIL_CLIENTE", email);
  setValor_(hoja, h, filaCliente, "ESTADO_CLIENTE", "ACTIVO");
  if (notas) setValor_(hoja, h, filaCliente, "NOTAS", notas);
  if (incrementarVisita) {
    const total = Number(hoja.getRange(filaCliente, h.indexOf("TOTAL_VISITAS") + 1).getValue()) || 0;
    setValor_(hoja, h, filaCliente, "ULTIMA_VISITA", fechaCita);
    setValor_(hoja, h, filaCliente, "PROXIMO_RECORDATORIO_CORTE", proximo);
    setValor_(hoja, h, filaCliente, "TOTAL_VISITAS", total + 1);
  }
}

/************ RESUMEN DIARIO BARBERO ************/

function enviarResumenDiarioBarbero() {
  if (!sistemaActivo_()) { registrarLog_("SISTEMA", "resumen_barbero", "", "PAUSADO"); return; }
  const config = cargarConfig_();
  const emailBarbero = String(config["EMAIL_BARBERO"] || "").trim();
  const nombreBarbero = String(config["NOMBRE_BARBERO"] || "Barbero").trim();
  if (!emailBarbero) return;

  const hoja = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(HOJAS.CITAS);
  if (!hoja) return;
  const data = hoja.getDataRange().getValues(); if (data.length <= 1) return;
  const headers = data[0]; const hoy = limpiarFecha_(new Date());

  const citasHoy = [];
  for (let i = 1; i < data.length; i++) {
    const fila = data[i];
    const fecha = limpiarFecha_(getValor_(fila, headers, "FECHA_CITA"));
    if (!fecha || fecha.getTime() !== hoy.getTime()) continue;
    const est = String(getValor_(fila, headers, "ESTADO_CITA")).toUpperCase();
    if (est === "CANCELADA" || est === "REAGENDADA") continue;
    citasHoy.push({
      hora: formatearHora_(getValor_(fila, headers, "HORA_CITA")),
      nombre: getValor_(fila, headers, "NOMBRE_CLIENTE"),
      servicio: getValor_(fila, headers, "SERVICIO"),
      telefono: getValor_(fila, headers, "TELEFONO")
    });
  }
  citasHoy.sort((a, b) => {
    const m = h => { const p = String(h).split(":"); return (Number(p[0]) || 0) * 60 + (Number(p[1]) || 0); };
    return m(a.hora) - m(b.hora);
  });

  const nombreBarberia = config["NOMBRE_BARBERIA"] || "Barber Studio";
  const acento = config["COLOR_ACENTO"] || "#1A1A1A";
  const logoUrl = String(config["LOGO_URL"] || "").trim();
  const fechaStr = (() => { const s = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "EEEE dd 'de' MMMM"); return s.charAt(0).toUpperCase() + s.slice(1); })();
  const asunto = citasHoy.length > 0 ? `📅 Tu agenda — ${fechaStr} (${citasHoy.length} cita${citasHoy.length > 1 ? "s" : ""})` : `📅 Tu agenda — ${fechaStr}`;

  let textoPlano = `Hola ${nombreBarbero}, tu agenda para hoy (${fechaStr}):\n\n`;
  citasHoy.length === 0 ? textoPlano += "No tienes citas. ¡Día libre!" : citasHoy.forEach(c => { textoPlano += `${c.hora}  ${c.nombre}  →  ${c.servicio}\n`; });

  try {
    MailApp.sendEmail({ to: emailBarbero, subject: asunto, body: textoPlano,
      htmlBody: construirEmailResumenBarbero_(citasHoy, nombreBarbero, nombreBarberia, acento, fechaStr, logoUrl), name: nombreBarberia });
    registrarLog_("RESUMEN_BARBERO", emailBarbero, `${citasHoy.length} citas`, "ENVIADO");
  } catch (e) { registrarLog_("ERROR_RESUMEN_BARBERO", emailBarbero, "", e.message); }
}

/************ ENVÍO DE MENSAJES (GATE DE PAUSA) ************/

function enviarMensajeCliente_(telefono, email, mensajeTexto, tipo, config, vars) {
  const cfg = config || cargarConfig_();
  if (!sistemaActivo_(cfg)) { registrarLog_(tipo, telefono || email || "", mensajeTexto, "PAUSADO"); return false; }
  if (esSi_(cfg["WHATSAPP_ACTIVO"])) {
    if (telefono) enviarWhatsAppTwilio_(telefono, tipo, vars || {}, cfg, mensajeTexto);
    else registrarLog_(tipo + "_WHATSAPP", "SIN TEL", mensajeTexto, "SIN TELEFONO");
  } else { registrarLog_(tipo + "_WHATSAPP", telefono || "", mensajeTexto, "WHATSAPP DESACTIVADO"); }
  if (esSi_(cfg["EMAIL_ACTIVO"]) && email) { enviarCorreoCliente_(email, mensajeTexto, tipo, cfg, vars || {}); }
  else { registrarLog_(tipo + "_EMAIL", email || "SIN EMAIL", mensajeTexto, esSi_(cfg["EMAIL_ACTIVO"]) ? "SIN EMAIL" : "EMAIL DESACTIVADO"); }
  return true;
}

/************ TWILIO — WHATSAPP ************/

function enviarWhatsAppTwilio_(telefono, tipo, vars, config, textoFallback) {
  try {
    const sid = String(config["TWILIO_ACCOUNT_SID"] || "").trim();
    const token = String(config["TWILIO_AUTH_TOKEN"] || "").trim();
    if (!sid || !token) { registrarLog_(tipo + "_WHATSAPP", telefono, "", "FALTAN CREDENCIALES TWILIO"); return; }
    const to = "whatsapp:" + normalizarTelefono_(telefono, config);
    const payload = { To: to };
    const msgService = String(config["TWILIO_MESSAGING_SERVICE_SID"] || "").trim();
    let from = String(config["TWILIO_WHATSAPP_FROM"] || "").trim();
    if (msgService) payload.MessagingServiceSid = msgService;
    else if (from) {
      from = from.replace(/^whatsapp:/i, "").trim();          // quita el prefijo si viene
      if (from.charAt(0) !== "+") from = "+" + from.replace(/[^\d]/g, "");  // asegura +E.164
      payload.From = "whatsapp:" + from;
    }
    else { registrarLog_(tipo + "_WHATSAPP", telefono, "", "FALTA FROM o MESSAGING_SERVICE_SID"); return; }

    const contentSid = String(config["CONTENTSID_" + tipo] || "").trim();
    if (contentSid) {
      payload.ContentSid = contentSid;
      const varsWA = Object.assign({}, vars, { TELEFONO: String(config["TELEFONO_BARBERIA"] || "").trim(), BARBERIA: String(config["NOMBRE_BARBERIA"] || "").trim() });
      const cv = construirContentVariables_(tipo, varsWA);
      if (cv) payload.ContentVariables = cv;
    } else { payload.Body = textoFallback || ""; }

    const url = "https://api.twilio.com/2010-04-01/Accounts/" + sid + "/Messages.json";
    const resp = UrlFetchApp.fetch(url, { method: "post", payload: payload, headers: { Authorization: "Basic " + Utilities.base64Encode(sid + ":" + token) }, muteHttpExceptions: true });
    const code = resp.getResponseCode(); const body = resp.getContentText();
    if (code >= 200 && code < 300) {
      let s = ""; try { s = JSON.parse(body).sid || ""; } catch (e) {}
      registrarLog_(tipo + "_WHATSAPP", to, textoFallback || contentSid, "ENVIADO " + (s || code));
    } else { registrarLog_(tipo + "_WHATSAPP", to, String(body).substring(0, 300), "ERROR " + code); }
  } catch (e) { registrarLog_("ERROR_WHATSAPP", telefono, tipo, e.message); }
}

function construirContentVariables_(tipo, vars) {
  const orden = ORDEN_VARIABLES_WA[tipo];
  if (!orden || orden.length === 0) return "";
  const obj = {};
  orden.forEach((clave, idx) => { obj[String(idx + 1)] = String((vars && vars[clave] != null) ? vars[clave] : ""); });
  return JSON.stringify(obj);
}

function telClave_(tel, config) {
  return normalizarTelefono_(tel, config || cargarConfig_());
}

function normalizarTelefono_(tel, config) {
  let s = String(tel || "").replace(/[^\d+]/g, "");
  if (!s) return "";
  if (s.charAt(0) === "+") return s;
  const lada = String((config && config["LADA_PAIS"]) || "52").replace(/\D/g, "") || "52";
  s = s.replace(/^0+/, "");
  if (s.indexOf("00") === 0) s = s.slice(2);
  if (s.indexOf(lada) === 0 && s.length > 10) return "+" + s;
  return "+" + lada + s;
}

function probarWhatsAppTwilio() {
  const ui = SpreadsheetApp.getUi(); const cfg = cargarConfig_();
  const resp = ui.prompt("🧪 Probar WhatsApp", "Número de destino (con lada, ej: 9671234567):", ui.ButtonSet.OK_CANCEL);

}

function construirMensajeDesdeCache_(plantillas, tipo, variables, config) {
  let msg = plantillas[tipo] || "";

  if (config) { extra.BARBERIA = config["NOMBRE_BARBERIA"] || ""; extra.TELEFONO = config["TELEFONO_BARBERIA"] || ""; extra.IG = config["INSTAGRAM_BARBERIA"] || ""; }

function esSi_(v) { const s = String(v == null ? "" : v).trim().toUpperCase(); return s === "SÍ" || s === "SI" || s === "YES" || s === "TRUE" || s === "1"; }

function esCitaCerrada_(estado) { const e = String(estado || "").trim().toUpperCase(); return e === "COMPLETADA" || e === "CANCELADA" || e === "REAGENDADA"; }

  const headers = hoja.getRange(1, 1, 1, hoja.getLastColumn()).getValues()[0];
  const fila = headers.map(h => { const k = String(h).trim(); return (obj[k] !== undefined && obj[k] !== null) ? obj[k] : ""; });

}

function escaparHtml_(s) {
  return String(s == null ? "" : s).replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;").replace(/"/g, "&quot;").replace(/'/g, "&#39;");
}

function crearId_(prefijo) { return `${prefijo}-${new Date().getTime()}-${Math.floor(Math.random() * 9999)}`; }

function limpiarFecha_(fecha) {
  if (!fecha) return null;
  const str = String(fecha).trim();
  const iso = str.match(/^(\d{4})[\/\-](\d{1,2})[\/\-](\d{1,2})$/);
  if (iso) { const f = new Date(Number(iso[1]), Number(iso[2]) - 1, Number(iso[3])); f.setHours(0,0,0,0); return isNaN(f.getTime()) ? null : f; }
  const dmy = str.match(/^(\d{1,2})[\/\-](\d{1,2})[\/\-](\d{4})$/);
  if (dmy) { const f = new Date(Number(dmy[3]), Number(dmy[2]) - 1, Number(dmy[1])); f.setHours(0,0,0,0); return isNaN(f.getTime()) ? null : f; }
  const f = new Date(fecha); if (isNaN(f.getTime())) return null; f.setHours(0,0,0,0); return f;
}

function formatearFecha_(fecha) {
  if (!fecha) return "";
  try { return Utilities.formatDate(new Date(fecha), Session.getScriptTimeZone(), "dd/MM/yyyy"); } catch (e) { return String(fecha); }
}

function formatearHora_(hora) {
  if (!hora) return "";
  if (hora instanceof Date) { try { return Utilities.formatDate(hora, Session.getScriptTimeZone(), "HH:mm"); } catch (e) { return String(hora); } }
  return String(hora);
}

function formatearFechaAmable_(fecha) {
  const f = limpiarFecha_(fecha);
  if (!f) return String(fecha || "");
  const dias = ["domingo","lunes","martes","miércoles","jueves","viernes","sábado"];
  const meses = ["enero","febrero","marzo","abril","mayo","junio","julio","agosto","septiembre","octubre","noviembre","diciembre"];
  return `${dias[f.getDay()]} ${f.getDate()} de ${meses[f.getMonth()]}`;
}

function formatearHoraAmable_(hora) {
  if (!hora) return "";
  let hh, mm;
  if (hora instanceof Date) { hh = hora.getHours(); mm = hora.getMinutes(); }
  else {
    const s = String(hora).trim(); const t = new Date(`1970-01-01 ${s}`);
    if (!isNaN(t.getTime())) { hh = t.getHours(); mm = t.getMinutes(); }
    else { const p = s.replace(/[^\d:]/g, "").split(":"); hh = Number(p[0]) || 0; mm = Number(p[1]) || 0; }
  }
  let parte;
  if (hh === 0) parte = "de la noche";
  else if (hh >= 1 && hh <= 4) parte = "de la madrugada";
  else if (hh >= 5 && hh <= 11) parte = "de la mañana";
  else if (hh === 12) parte = "del mediodía";
  else if (hh >= 13 && hh <= 18) parte = "de la tarde";
  else parte = "de la noche";
  let h12 = hh % 12; if (h12 === 0) h12 = 12;
  const minStr = (mm === 0) ? "" : ":" + (mm < 10 ? "0" + mm : mm);
  return `${h12}${minStr} ${parte}`;
}
